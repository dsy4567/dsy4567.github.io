    /**
 * 计算平均分
 * @param {number[]} 分数数组 分数列表
 * @returns {number} 平均分
 */
function 计算平均分(分数数组) {
    // 制表符缩进，与站点代码风格一致
    const 总分 = 分数数组.reduce((累加, 当前) => 累加 + 当前, 0);
    return 总分 / 分数数组.length;
}
